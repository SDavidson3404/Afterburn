@tool
extends EditorScript

func _run() -> void:
	var output_file = FileAccess.open("res://all_project_code.txt", FileAccess.WRITE)
	if output_file:
		_combine_directory("res://", output_file)
		output_file.close()
		print("SUCCESS: Created all_project_code.txt with scripts AND scenes!")

func _combine_directory(path: String, output_file: FileAccess) -> void:
	var dir = DirAccess.open(path)
	if not dir:
		return
		
	dir.list_dir_begin()
	var file_name = dir.get_next()
	
	while file_name != "":
		# Ignore hidden folders like .godot, .import, and addons
		if dir.current_is_dir():
			if not file_name.begins_with(".") and file_name != "addons":
				_combine_directory(path + file_name + "/", output_file)
		else:
			# Includes both .gd scripts and .tscn scene text files
			if file_name.ends_with(".gd") or file_name.ends_with(".tscn"):
				var file_path = path + file_name
				var script_file = FileAccess.open(file_path, FileAccess.READ)
				if script_file:
					output_file.store_string("\n\n" + "=".repeat(80) + "\n")
					output_file.store_string("FILE: " + file_path + "\n")
					output_file.store_string("=".repeat(80) + "\n\n")
					output_file.store_string(script_file.get_as_text())
					script_file.close()
					
		file_name = dir.get_next()

## Dedicated fuel crystal spawner.
## Manages speed-adaptive vector projection, reaction-time spawn pacing, spatial miss detection, and collectible lifecycle management.
extends Node3D

@export_category("Node References")
## PackedScene instantiated for crystal collectibles.
@export var crystal_scene: PackedScene

## Reference to the main player node or character body.
@export var player: Node3D

## Reference to the main power manager node.
@export var power: Node

@export_category("Fuel Drain & Timing")
## Minimum baseline fuel drain window in seconds before spawning a crystal.
@export var min_drain_time: float = 4.0

## Maximum baseline fuel drain window in seconds before spawning a crystal.
@export var max_drain_time: float = 5.0

@export_category("Lanes Setup")
## Horizontal width of each movement lane in world units.
@export var lane_width: float = 3.0

## Total number of track movement lanes across the street.
@export var total_lanes: int = 5

# Runtime spawner state tracking
var _active_crystal: Node3D = null
var _spawn_timer: float = 0.0
var _is_waiting_to_spawn: bool = false


func _ready() -> void:
	_calculate_and_start_spawn_timer()


## Safely extracts the player's physical CharacterBody3D node or returns the root player node.
func _get_player_body() -> Node3D:
	if not player:
		return null
	var body_node = player.get("body")
	return body_node as Node3D if body_node else player


func _process(delta: float) -> void:
	if not power or not power.get("is_run_active"):
		return

	# Monitor active crystal position relative to player to detect missed collectibles
	if is_instance_valid(_active_crystal):
		var target_body := _get_player_body()
		if target_body:
			var relative_z: float = target_body.to_local(_active_crystal.global_position).z
			# Trigger miss handler if crystal moves 5+ units behind the player
			if relative_z > 5.0:
				on_crystal_missed()
		return

	# Countdown spawn timer when waiting to instantiate the next crystal
	if _is_waiting_to_spawn:
		_spawn_timer -= delta
		if _spawn_timer <= 0.0:
			_is_waiting_to_spawn = false
			spawn_crystal()


## Projects and instantiates a new fuel crystal ahead of the player in 3D space.
func spawn_crystal() -> void:
	if not crystal_scene or is_instance_valid(_active_crystal):
		return

	var target_body := _get_player_body()
	if not target_body:
		return

	var crystal = crystal_scene.instantiate() as Node3D
	get_parent().add_child(crystal)
	_active_crystal = crystal

	# Pass spawner reference to collectible for pickup callback management
	if crystal.has_method("set_spawner"):
		crystal.set_spawner(self)

	# Query dynamic spawn distance scaled to current forward speed
	var spawn_distance: float = power.get_current_spawn_distance() if power and power.has_method("get_current_spawn_distance") else 45.0

	# Pick a random lane index and convert to symmetric X-coordinate offset
	var random_lane: int = randi() % total_lanes
	@warning_ignore("integer_division")
	var lane_offset_index: int = random_lane - (total_lanes / 2)
	var x_offset: float = lane_offset_index * lane_width

	# Project position forward along player transform basis
	var spawn_pos: Vector3 = target_body.global_position
	spawn_pos += -target_body.global_transform.basis.z * spawn_distance
	spawn_pos.x = x_offset
	spawn_pos.y = randf_range(1.5, 2.5)

	crystal.global_position = spawn_pos


## Callback triggered when player collects the active crystal.
func on_crystal_collected() -> void:
	_clear_active_crystal()
	if power:
		power.collect_crystal()
	if get_node_or_null("/root/DatabaseManager"):
		DatabaseManager.add_crystal()
	_calculate_and_start_spawn_timer()


## Callback triggered when active crystal passes behind player uncollected.
func on_crystal_missed() -> void:
	_clear_active_crystal()
	_calculate_and_start_spawn_timer()


## Frees active crystal node from memory and clears local reference.
func _clear_active_crystal() -> void:
	if is_instance_valid(_active_crystal):
		_active_crystal.queue_free()
		_active_crystal = null


## Calculates spawn timer delay factoring in random drain bounds, invincibility, and reaction time.
func _calculate_and_start_spawn_timer() -> void:
	var target_drain_time: float = randf_range(min_drain_time, max_drain_time)
	var active_invincibility: float = 0.0
	if power and power.get("is_invincible"):
		active_invincibility = power.get("invincibility_timer")

	var reaction_time: float = power.get_reaction_time() if power and power.has_method("get_reaction_time") else 3.0

	# Timer formula: active_invincibility + target_drain_time - reaction_time
	_spawn_timer = maxf(0.05, active_invincibility + target_drain_time - reaction_time)
	_is_waiting_to_spawn = true

## Singleton database manager handling real-time run telemetry, score calculations, SQLite data persistence, and aggregate lifetime analytics.
extends Node

## SQLite database instance reference.
var db: SQLite

# Distance scoring constants and runtime trackers
const POINTS_PER_METER: float = 1.0
var total_distance_meters: int = 0
var meters_traveled: float = 0.0
var travel_score: int = 0

# Dodge scoring constants and runtime trackers
const POINTS_PER_DODGE: int = 10
var dodge_score: int = 0

# Crystal scoring constants and runtime trackers
const CRYSTAL_POINTS: int = 100
var crystal_score: int = 0

# Stat tracking counters logged during each run
var power_crystals: int = 0
var gadgets: int = 0
var jump_obstacle: int = 0
var slide_obstacle: int = 0
var dash_obstacle: int = 0
var robot_crash: int = 0
var scaffolding_crash: int = 0
var drone_crash: int = 0


## Initializes SQLite database connection and creates required tables upon engine launch.
func _ready() -> void:
	db = SQLite.new()
	db.path = "user://game_data.db"
	db.open_db()
	
	create_stats_table()


## Executes SQL DDL query to establish the stats table schema if missing.
func create_stats_table() -> void:
	var stats_query := """
	CREATE TABLE IF NOT EXISTS stats (
		stats_id INTEGER PRIMARY KEY,
		total_distance INTEGER,
		travel_score INTEGER,
		dodge_score INTEGER,
		crystal_score INTEGER,
		total_score INTEGER,
		power_crystals INTEGER,
		gadgets INTEGER,
		jump_obstacle INTEGER,
		slide_obstacle INTEGER,
		dash_obstacle INTEGER,
		robot_crash INTEGER,
		scaffolding_crash INTEGER,
		drone_crash INTEGER
	);
	"""
	db.query(stats_query)


## Tracks distance moved during an active run and calculates corresponding travel score points.
func add_distance(meters_moved: float) -> void:
	meters_traveled += meters_moved
	total_distance_meters = int(meters_traveled)
	travel_score = int(total_distance_meters * POINTS_PER_METER)


## Logs successful obstacle dodges by maneuver type and increments dodge score.
func add_dodge(dodge_type: String = "") -> void:
	dodge_score += POINTS_PER_DODGE
	if dodge_type == "jump":
		jump_obstacle += 1
	elif dodge_type == "slide":
		slide_obstacle += 1
	elif dodge_type == "dash":
		dash_obstacle += 1


## Logs collected power crystals and updates crystal score total.
func add_crystal() -> void:
	crystal_score += CRYSTAL_POINTS
	power_crystals += 1


## Increments total gadget pickup count.
func add_gadget() -> void:
	gadgets += 1


## Returns total combined score across movement distance, dodges, and crystals.
func display_score() -> int:
	return (travel_score + dodge_score + crystal_score)


## Logs obstacle collisions categorized by obstacle type.
func player_crash(crash_type: String = "") -> void:
	if crash_type == "robot":
		robot_crash += 1
	elif crash_type == "scaffolding":
		scaffolding_crash += 1
	elif crash_type == "drone":
		drone_crash += 1


## Inserts current run statistics into SQLite database and returns summary dictionary.
func save_run() -> Dictionary:
	var run_data: Dictionary = {
		"total_distance": total_distance_meters,
		"travel_score": travel_score,
		"dodge_score": dodge_score,
		"crystal_score": crystal_score,
		"total_score": display_score(),
		"power_crystals": power_crystals,
		"gadgets": gadgets,
		"jump_obstacle": jump_obstacle,
		"slide_obstacle": slide_obstacle,
		"dash_obstacle": dash_obstacle,
		"robot_crash": robot_crash,
		"scaffolding_crash": scaffolding_crash,
		"drone_crash": drone_crash
	}
	db.insert_row("stats", run_data)
	return run_data


## Zeroes active run statistics counters in preparation for a new run.
func reset_run_stats() -> void:
	total_distance_meters = 0
	meters_traveled = 0.0
	travel_score = 0
	dodge_score = 0
	crystal_score = 0
	power_crystals = 0
	gadgets = 0
	jump_obstacle = 0
	slide_obstacle = 0
	dash_obstacle = 0
	robot_crash = 0
	scaffolding_crash = 0
	drone_crash = 0


## Queries database for high scores and aggregated lifetime player analytics.
func lifetime_stats() -> Dictionary:
	var lifetime_query: String = """
	SELECT
		COUNT(*) AS total_runs,
		COALESCE(MAX(total_score), 0) AS high_score,
		COALESCE(MAX(total_distance), 0) AS max_distance,
		COALESCE(MAX(power_crystals), 0) AS max_crystals,
		COALESCE(MAX(jump_obstacle + slide_obstacle + dash_obstacle), 0) AS max_dodges,
		COALESCE(SUM(total_distance), 0) AS total_distance,
		COALESCE(SUM(power_crystals), 0) AS total_crystals,
		COALESCE(SUM(gadgets), 0) AS total_gadgets,
		COALESCE(SUM(jump_obstacle + slide_obstacle + dash_obstacle), 0) AS total_dodges,
		COALESCE(SUM(robot_crash), 0) AS robot_crashes,
		COALESCE(SUM(scaffolding_crash), 0) AS scaffolding_crashes,
		COALESCE(SUM(drone_crash), 0) AS drone_crashes
	FROM stats;
	"""
	# COALESCE(..., 0) prevents NULL returns when database table contains no records yet
	db.query(lifetime_query)
	if db.query_result.size() > 0:
		return db.query_result[0]
	return {}

## World interactable node representing collectible gadget pickups on the track.
## Handles dynamic visual coloring, collision detection, and instant ability activation.
extends Area3D

## Available gadget types selectable in the editor or randomly rolled at runtime.
enum GadgetType { RANDOM, PULSE_WAVE, MAGNET, POWER_ARMOR, HOVERBOARD, OVERDRIVEN_ROCKET }

@export_category("Pickup Configuration")
## The specific gadget type this node represents. Defaults to RANDOM for procedural spawns.
@export var gadget_type: GadgetType = GadgetType.RANDOM

## Guard flag preventing duplicate pickup activations within the same frame.
var _is_collected: bool = false

## Maps inspectable enum values to string keys required by the GadgetBase factory.
const TYPE_TO_ID: Dictionary = {
	GadgetType.PULSE_WAVE: "pulse_wave",
	GadgetType.MAGNET: "magnet",
	GadgetType.POWER_ARMOR: "power_armor",
	GadgetType.HOVERBOARD: "hoverboard",
	GadgetType.OVERDRIVEN_ROCKET: "overdriven_rocket"
}


func _ready() -> void:
	# Group assignment enables magnetic attraction by the Magnet gadget
	add_to_group("Collectibles")
	body_entered.connect(_on_body_entered)
	_resolve_gadget_type_and_color()


## Resolves random pickup types and sets up color-coded emissive materials.
func _resolve_gadget_type_and_color() -> void:
	if gadget_type == GadgetType.RANDOM:
		gadget_type = [
			GadgetType.PULSE_WAVE,
			GadgetType.MAGNET,
			GadgetType.POWER_ARMOR,
			GadgetType.HOVERBOARD,
			GadgetType.OVERDRIVEN_ROCKET
		].pick_random()

	_apply_gadget_color()


## Programmatically overrides the mesh material to color-code the pickup in 3D space.
func _apply_gadget_color() -> void:
	var mesh_node := get_node_or_null("MeshInstance3D") as MeshInstance3D
	if not mesh_node:
		return

	var gadget_color: Color = Color.WHITE
	match gadget_type:
		GadgetType.PULSE_WAVE: gadget_color = Color.MAGENTA
		GadgetType.MAGNET: gadget_color = Color.RED
		GadgetType.POWER_ARMOR: gadget_color = Color.GREEN
		GadgetType.HOVERBOARD: gadget_color = Color.WHITE
		GadgetType.OVERDRIVEN_ROCKET: gadget_color = Color(1.0, 0.35, 0.0)

	var material := StandardMaterial3D.new()
	material.albedo_color = gadget_color
	material.emission_enabled = true
	material.emission = gadget_color
	material.emission_energy_multiplier = 1.5

	mesh_node.material_override = material


## Handles player intersection, triggering ability effects and freeing the pickup node.
func _on_body_entered(body: Node3D) -> void:
	if _is_collected:
		return

	# Support collisions on either the root player node or sub-character body
	var player_node: Node3D = body if body is Player else body.get_parent() as Node3D
	if player_node and player_node.is_in_group("Player"):
		_is_collected = true
		_trigger_pickup_effect(player_node)
		queue_free()


## Uses the GadgetBase factory to instantiate and immediately activate the gadget power.
func _trigger_pickup_effect(player: Node3D) -> void:
	var power_mgr = player.get("power_node") as Node
	if not power_mgr:
		return

	var gadget_id: String = TYPE_TO_ID.get(gadget_type, "")
	var gadget_instance: GadgetBase = GadgetBase.create_by_id(gadget_id)
	if gadget_instance:
		gadget_instance.activate_ability(player, power_mgr)
		# Free temporary factory instance to prevent memory leaks
		gadget_instance.free()

## Dedicated spawner node that projects standalone gadget pickups ahead of the player at timed intervals.
## Operates independently of track generation to ensure consistent gadget availability.
extends Node3D

@export_category("Scene References")
## PackedScene file instantiated when spawning a gadget pickup in 3D space.
@export var gadget_pickup_scene: PackedScene

## Reference to the main player node. Auto-assigned at runtime if left null.
@export var player: Player

@export_category("Spawn Timing")
## Minimum interval in seconds between periodic gadget spawns.
@export var min_spawn_interval: float = 20.0

## Maximum interval in seconds between periodic gadget spawns.
@export var max_spawn_interval: float = 25.0

@export_category("Track Layout")
## Total number of active movement lanes across the track.
@export var total_lanes: int = 5

## Width of each lane in world units.
@export var lane_width: float = 3.0

## Distance ahead of the player (in Z-axis world units) where pickups are spawned.
@export var spawn_distance_ahead: float = 100.0

## Internal countdown timer tracking seconds remaining until the next spawn attempt.
var _spawn_timer: float = 0.0


func _ready() -> void:
	if not player:
		player = get_tree().get_first_node_in_group("Player") as Player
	
	# Set initial delay to a random float between 20.0 and 25.0 seconds
	_spawn_timer = randf_range(min_spawn_interval, max_spawn_interval)


func _process(delta: float) -> void:
	if not gadget_pickup_scene:
		return

	# Pause timer countdown when run is inactive (menu screens or game over)
	var target_player := _get_moving_target()
	if target_player:
		var power_mgr = target_player.get("power_node") as Node
		if power_mgr and not power_mgr.get("is_run_active"):
			return

	_spawn_timer -= delta
	if _spawn_timer <= 0.0:
		_spawn_gadget()
		_spawn_timer = randf_range(min_spawn_interval, max_spawn_interval)


## Calculates track position and instantiates a gadget pickup ahead of the player.
func _spawn_gadget() -> void:
	var target_body: Node3D = _get_moving_target()
	if not target_body:
		return

	# Pick a random lane index and convert it to a symmetric X-offset centered at 0.0
	var lane: int = randi() % total_lanes
	@warning_ignore("integer_division")
	var lane_offset: int = lane - (total_lanes / 2)
	var gadget_x: float = lane_offset * lane_width

	var gadget_instance = gadget_pickup_scene.instantiate() as Node3D
	get_parent().add_child(gadget_instance)

	# Position pickup floating 1.0 unit above the track floor at the calculated lane and Z-distance
	var spawn_z: float = target_body.global_position.z - spawn_distance_ahead
	gadget_instance.global_position = Vector3(gadget_x, 1.0, spawn_z)


## Safely extracts the player's physical body node or falls back to the player root node.
func _get_moving_target() -> Node3D:
	if not player:
		player = get_tree().get_first_node_in_group("Player") as Player

	if player and player.body:
		return player.body

	return player

## Hoverboard loadout gadget.
## Trades accelerated track speed progression for boosted jump velocity, reduced dash drain, and extended coasting.
extends GadgetBase

@export_category("Ability Properties")
## Total active duration of the hoverboard buff in seconds.
@export var duration: float = 20.0


func _init() -> void:
	gadget_name = "Hoverboard"
	gadget_id = "hoverboard"


## Applies loadout handicap: Increases overall run acceleration rate by 15%.
func apply_loadout_nerf(power_manager: Node) -> void:
	power_manager.speed_acceleration *= 1.15


## Triggers hoverboard active state or refreshes duration if already active.
func activate_ability(player: Node3D, power_manager: Node) -> void:

## Inner component node attached to the player that pulls track collectibles dynamically.
class MagnetInstance extends Node:
	## Active duration remaining in seconds.
	var duration: float = 15.0
	
	## Max duration cap used for timer refreshes.
	var max_duration: float = 15.0
	
	## Reference to the target player node.
	var player: Node3D = null
	
	## Maximum lateral/vertical cross-section distance (in units) for item attraction.
	const MAX_XY_RADIUS: float = 3.5


	func _physics_process(delta: float) -> void:
		duration -= delta
		if duration <= 0.0:
			queue_free()
			return

		if not is_instance_valid(player):
			return

		var player_node := player as Player
		var player_pos: Vector3 = player_node.body.global_position if player_node and player_node.body else player.global_position
		var current_spd: float = player_node.current_speed if player_node else 15.0

		# Dynamically scale Z-detection range and pull speed based on forward velocity
		var dynamic_z_window: float = maxf(1.5, current_spd * 0.05)
		var dynamic_pull_speed: float = maxf(120.0, current_spd * 2.5)

		# Iterate through all nodes registered under the Collectibles group
		for item in player.get_tree().get_nodes_in_group("Collectibles"):
			if is_instance_valid(item) and item is Node3D:
				var delta_x: float = item.global_position.x - player_pos.x
				var delta_y: float = item.global_position.y - player_pos.y
				var xy_distance: float = Vector2(delta_x, delta_y).length()
				var z_distance: float = absf(item.global_position.z - player_pos.z)

				# Pull collectible toward player if within lateral radius and Z-axis window
				if xy_distance <= MAX_XY_RADIUS and z_distance <= dynamic_z_window:
					item.global_position.x = move_toward(item.global_position.x, player_pos.x, dynamic_pull_speed * delta)
					item.global_position.y = move_toward(item.global_position.y, player_pos.y, dynamic_pull_speed * delta)

		return

	if model and model.has_method("play_anim") and not animation_player.is_playing():
		model.play_anim("Idle")
	# Set the power meter to reflect the current power
	if power_meter:
		power_meter.value = min(power_node.current_power, 100.0)
	
	# Update text to reflect current speed and score

	score_label.text = ("SCORE: " + str(DatabaseManager.display_score()))
	
	# Update the row squares
	update_row_square()
	
	# Set sideways movement to 0
	body.velocity.x = 0.0
	
	# If you are dashing, set the velocity to current speed of the power, 
	# times the dash multiplier and set is dashing to false
	if is_dashing:
		body.velocity.z = -power_node.current_speed * dash_multiplier
		is_dashing = false
	
	# Otherwise if you are starting the run, set the initial velocity and note that it was set up
	elif starting_run:
		body.velocity.z = -power_node.current_speed
		starting_run = false
	
	# If neither are true, change the speed towards the current speed in power by 10 MPS/s
	else:
		var target_vel_z: float = -power_node.current_speed
		if body.velocity.z < target_vel_z:
			var active_drain_rate: float = DASH_DRAIN_RATE * dash_drain_multiplier
			body.velocity.z = move_toward(body.velocity.z, target_vel_z, active_drain_rate * delta)
		else:
			body.velocity.z = move_toward(body.velocity.z, target_vel_z, 50.0 * delta)
	
	# Set the current speed to the forward velocity of the body
	current_speed = -body.velocity.z
	
	# move position of player to the current row and bring rows with body
	body.global_position.x = move_toward(body.global_position.x, current_row_offset, STRAFE_SPEED * delta)
	
	# Runs if space is pressed and you are on the floor
	if Input.is_action_just_pressed("Space") and body.is_on_floor():
		
		# If you are sliding, cancel it
		if is_sliding:
			if animation_player.current_animation == "Slide":

		# Apply jump velocity
		body.velocity.y += (JUMP_VELOCITY * jump_velocity_multiplier)
	
	# If player is not on floor, apply gravity
	if not body.is_on_floor():
		body.velocity.y -= (GRAVITY * gravity_multiplier) * delta
	
	# Move the body according to the set velocities
	body.move_and_slide()

## Central run state manager controlling player power economy, kinematic speed scaling, damage resolution pipelines, and gadget cooldown lifecycles.
extends Node

## Emitted when loadout gadget cooldown status or progress percentage updates.

## Emitted when a run terminates, triggering game over UI and data saving.

@export_category("Run Configuration")
## Maximum suit power meter capacity.
@export var max_power: float = 100.0

## Passive suit power drain rate per second during normal operation.
@export var power_drain_rate: float = 5.0

## Baseline forward movement speed at run start in meters per second.
@export var base_speed: float = 15.0

## Terminal forward movement velocity cap (Mach 1 equivalent in m/s).
@export var max_speed: float = 343.0

## Acceleration constant used in kinematic speed scaling calculations.
@export var speed_acceleration: float = 0.25

## Rate at which forward speed decays per second when power reaches zero.
@export var coasting_deceleration: float = 5.0

@export var crystal_invincibility_duration: float = 3.0

## Suit power cost subtracted when executing a slide dodge maneuver.
@export var slide_cost: int = 10

## Suit power cost subtracted when executing a dash speed boost.
@export var dash_cost: int = 25

## Reference to the active Player node.
@export var player: Player

@export_category("Gadget Setup")
## Active GadgetBase instance equipped for the current run.
@export var selected_gadget: GadgetBase = null

var passive_drain_multiplier: float = 1.0

var is_invincible: bool = false

## Resets all run variables, applies loadout gadget nerfs, and starts active gameplay.
func start_run() -> void:
	if get_node_or_null("/root/DatabaseManager"):
		DatabaseManager.reset_run_stats()

	# Reset suit modifiers and overdrive states to baseline defaults
	is_overcharged = false
	overdrive_rocket_timer = 0.0
	max_power = 100.0
	crystal_power_yield = 1.0
	passive_drain_multiplier = 1.0
	obstacle_damage_multiplier = 1.0
	coasting_deceleration_multiplier = 1.0
	has_overdriven_rocket = false
	is_overdrive_surge = false
	active_armor_layers.clear()
	
	# Apply loadout handicaps and initialize cooldown state if a gadget is equipped
	if selected_gadget:
		selected_gadget.apply_loadout_nerf(self)
		gadget_cooldown_progress = 0.0
		is_gadget_ready = true
		gadget_cooldown_changed.emit(0.0, true)
		
	current_power = max_power
	current_speed = base_speed
	target_speed = base_speed
	distance_traveled = 0.0
	is_run_active = true
	is_coasting = false
	is_invincible = false
	invincibility_timer = 0.0
	speed_recovery_timer = 0.0
	
	power_changed.emit(current_power, max_power)
	
	# Bind player action signals to power drain callbacks
	if player:
		if not player.sliding.is_connected(slide):
			player.sliding.connect(slide)
		if not player.dashing.is_connected(dash):
			player.dashing.connect(dash)


func _process(delta: float) -> void:
	if not is_run_active:
		return
		
	var was_invincible_this_frame: bool = is_invincible
	
	# Process invincibility expiration and speed catch-up logic
	if is_invincible:
		var speed_ratio: float = current_speed / target_speed if target_speed > 0.0 else 1.0
		if speed_ratio >= 0.80:
			_set_invincible(false)
		else:
			invincibility_timer -= delta
			if invincibility_timer <= 0.0:
				_set_invincible(false)
			
	if speed_recovery_timer > 0.0:
		speed_recovery_timer = maxf(0.0, speed_recovery_timer - delta)
	
	# Track physical distance moved and update travel score
	var meters_moved: float = current_speed * delta
	distance_traveled += meters_moved
	
	if get_node_or_null("/root/DatabaseManager"):
		DatabaseManager.add_distance(meters_moved)
	
	# Kinematic target speed scaling equation: target_speed = sqrt(v_0^2 + 2 * a * d)
	target_speed = minf(sqrt((base_speed * base_speed) + (2.0 * speed_acceleration * distance_traveled)), max_speed)
	
	# State Machine Processing: Coasting vs Overdrive Surge vs Normal Operation
	if is_coasting:
		var active_coasting_rate: float = coasting_deceleration * coasting_deceleration_multiplier
		current_speed = maxf(0.0, current_speed - (active_coasting_rate * delta))
		if current_speed <= 0.0 and (not player or player.current_speed <= 0.0):
			end_run()
			
	elif is_overdrive_surge:
		# Overdrive Surge rapidly accelerates player toward top speed when fuel hits zero
		current_speed = move_toward(current_speed, max_speed, 50.0 * delta)
	
	else:
		if is_overcharged:
			# Rocket Overcharge state bypasses standard suit fuel drain
			var active_drain: float = power_drain_rate * passive_drain_multiplier
			overdrive_rocket_timer = maxf(0.0, overdrive_rocket_timer - (active_drain * delta))
			if overdrive_rocket_timer <= 0.0:
				is_overcharged = false
		else:
			if not was_invincible_this_frame:
				var active_drain: float = power_drain_rate * passive_drain_multiplier
				current_power = maxf(0.0, current_power - (active_drain * delta))
				power_changed.emit(current_power, max_power)
				
				if current_power <= 0.0:
					_handle_zero_power()
		
		# Catch-up acceleration logic restoring player speed after obstacle collisions
		if current_speed < target_speed:
			if speed_recovery_timer > 0.0:
				var speed_gap: float = target_speed - current_speed
				var required_acceleration: float = speed_gap / maxf(speed_recovery_timer, delta)
				current_speed = minf(target_speed, current_speed + (required_acceleration * delta))
			else:
				var fallback_acceleration: float = maxf(1.0, speed_acceleration) * 10.0
				current_speed = minf(target_speed, current_speed + (fallback_acceleration * delta))
		else:
			current_speed = target_speed
			speed_recovery_timer = 0.0
			if is_invincible:
				_set_invincible(false)


## Triggers power refill, restores coasting/surge states, and processes gadget cooldown reduction.
func collect_crystal() -> void:
	if not is_run_active:
		return
	
	current_power = clampf(current_power + (100.0 * crystal_power_yield), 0.0, max_power)
	
	if is_overcharged and overdrive_rocket_timer > 0.0:
		overdrive_rocket_timer = clampf(overdrive_rocket_timer + (overdrive_rocket_max_duration * crystal_power_yield), 0.0, overdrive_rocket_max_duration)
	
	if is_overdrive_surge:
		is_overdrive_surge = false
		current_speed = target_speed
	is_coasting = false
	
	# Grant catch-up invincibility if collecting a crystal while running below target speed
	var speed_ratio: float = current_speed / target_speed if target_speed > 0.0 else 1.0
	if current_speed < target_speed:
		if not has_overdriven_rocket and speed_ratio < 0.80:
			_set_invincible(true, crystal_invincibility_duration)
		else:
			_set_invincible(false)
		speed_recovery_timer = crystal_invincibility_duration
	else:
		_set_invincible(false)
		speed_recovery_timer = 0.0
		
	power_changed.emit(current_power, max_power)
	
	# Reduce equipped gadget cooldown if player is running at target speed without an active ability running
	if selected_gadget and not is_gadget_ready and is_at_target_speed():
		if _is_selected_gadget_active():
			return

		reduce_gadget_cooldown(1.0 / maxf(1.0, float(selected_gadget.cooldown_crystals_required)))


## Polymorphic helper querying whether the equipped gadget currently has an active state/buff.
func _is_selected_gadget_active() -> bool:
	if not selected_gadget:
		return false
	
	if selected_gadget.has_method("is_active"):
		return selected_gadget.is_active(player, self)
	
	match selected_gadget.gadget_id:
		"overdriven_rocket":
			return is_overcharged
		"power_armor":
			return active_armor_layers.size() > 0
		"hoverboard":
			return player != null and player.has_node("HoverboardInstance")
		"magnet":
			return player != null and player.has_node("MagnetInstance")
			
	return false


## Toggles invincibility status and notifies connected listeners via signals.
func _set_invincible(enabled: bool, duration: float = 0.0) -> void:
	is_invincible = enabled
	invincibility_timer = duration if enabled else 0.0
	invincibility_changed.emit(enabled)


## Handles zero-power fuel exhaustion, triggering either Overdrive Surge or standard coasting.
func _handle_zero_power() -> void:
	if has_overdriven_rocket:
		is_overdrive_surge = true
		if player and player.current_speed > current_speed:
			current_speed = player.current_speed
	else:
		is_coasting = true


## Reduces gadget cooldown progress ratio and emits UI update signals.
func reduce_gadget_cooldown(amount: float) -> void:
	gadget_cooldown_progress = maxf(0.0, gadget_cooldown_progress - amount)
	if gadget_cooldown_progress <= 0.001:
		is_gadget_ready = true
		gadget_cooldown_progress = 0.0
	gadget_cooldown_changed.emit(gadget_cooldown_progress, is_gadget_ready)


## Polymorphically triggers equipped gadget active power if ready and sets initial cooldown state.
func try_use_loadout_gadget() -> void:
	if not selected_gadget or not is_run_active or not is_gadget_ready or not player:
		return

	selected_gadget.activate_ability(player, self)
		
	is_gadget_ready = false
	if selected_gadget.gadget_id != "power_armor":
		gadget_cooldown_progress = 1.0
		gadget_cooldown_changed.emit(1.0, false)


## Engages overcharged rocket suit state for maximum active duration.
func activate_overdrive_rocket() -> void:
	is_overcharged = true
	overdrive_rocket_timer = overdrive_rocket_max_duration
	if is_overdrive_surge or is_coasting:
		is_overdrive_surge = false
		is_coasting = false
		if current_speed < target_speed:
			speed_recovery_timer = crystal_invincibility_duration
	power_changed.emit(current_power, max_power)


## Returns true if current forward speed matches current target speed threshold.
func is_at_target_speed() -> bool:
	return absf(current_speed - target_speed) < 0.5


## Calculates required player reaction window (3.0s down to 1.0s) based on target speed.
func get_reaction_time() -> float:
	var clamped_spd: float = clampf(target_speed, base_speed, max_speed)
	return remap(clamped_spd, base_speed, max_speed, 3.00, 1.00)


## Returns obstacle spawn distance ahead of player scaled to target velocity.
func get_current_spawn_distance() -> float:
	var clamped_spd: float = clampf(target_speed, base_speed, max_speed)
	return clamped_spd * get_reaction_time()


## Registers active Power Armor shield layer component with power manager.
func register_armor_layer(armor_node: Node) -> void:
	active_armor_layers.append(armor_node)


## Unregisters broken Power Armor shield layer and puts armor gadget back on cooldown.
func unregister_armor_layer(armor_node: Node) -> void:
	active_armor_layers.erase(armor_node)
	if selected_gadget and selected_gadget.gadget_id == "power_armor":
		is_gadget_ready = false
		gadget_cooldown_progress = 1.0
		gadget_cooldown_changed.emit(1.0, false)


## Intercepts collision damage through active armor, overdrive rocket, or suit power meter.
func obstacle_hit(damage: float = 25.0) -> void:
	if not is_run_active or is_invincible:
		return

	# Intercept hit if Power Armor shield layer is active
	if active_armor_layers.size() > 0:
		active_armor_layers[0].absorb_hit()
		return
		
	# Zero-power coasting/surge states result in immediate game over on hit
	if is_coasting or is_overdrive_surge:
		end_run()
		return
	
	var total_damage: float = damage * obstacle_damage_multiplier

	# Deduct damage from overcharged rocket duration before affecting main power meter
	if is_overcharged:
		if overdrive_rocket_timer > total_damage:
			overdrive_rocket_timer -= total_damage
			return
		else:
			total_damage -= overdrive_rocket_timer
			overdrive_rocket_timer = 0.0
			is_overcharged = false

	current_power = maxf(0.0, current_power - total_damage)
	power_changed.emit(current_power, max_power)
	if current_power <= 0.0:

func _process(_delta: float) -> void:
	if not is_instance_valid(player) or not player.body:
		return

	var player_z: float = player.body.global_position.z

	# Obstacle queue advancing loop: Flushes all passed rows even during high-speed frame spikes
	while not current_rows.is_empty():
		var front_row_z: float = current_rows.front().global_position.z
		if player_z < (front_row_z - ROW_DESPAWN_OFFSET):
			_advance_row_queue()
		else:
			break

	# Seamless road recycling: Trigger after player is 40m past road_2 start
	if is_instance_valid(road_2) and player_z < (road_2.global_position.z - ROAD_DESPAWN_OFFSET):
		advance_road_queue()

	road_1 = road_2
	road_2 = road_3
	road_3 = create_next_road(road_2)

## Summons a row of obstacles
func summon_row_obstacles() -> void:
	var taken_rows: Array = []
	var wall_spawns: Array = []
	var wall_count: int = 0
	
	row_count += 1
	var row_reference = Node3D.new()
	row_reference.name = "Row" + str(row_count)
	obstacles.add_child(row_reference)

	var basis_node: Node3D = player if not set_up else current_rows.back()
	var basis_z: float = basis_node.global_position.z if is_instance_valid(basis_node) else 0.0
	var spawn_dist: float = find_optimal_distance()
	var row_z: float = basis_z - spawn_dist

	row_reference.global_position = Vector3(0.0, 0.0, row_z)
	current_rows.append(row_reference)
	set_up = true
	
	# For loop for each row index
	for num in [-2, -1, 0, 1, 2]:
		if randi_range(1, 100) <= percentage_obstacle_spawned:
			if taken_rows.size() >= 5:
				break
			
			var obstacle_packed = obstacles_to_spawn.pick_random()
			var obstacle_instance = obstacle_packed.instantiate() as Obstacle
			
			var available_rows: Array = []
			for r in obstacle_instance.possible_main_rows:
				if not (r in taken_rows):
					available_rows.append(r)
			
			if available_rows.is_empty():
				obstacle_instance.queue_free()
				continue
			
			row_reference.add_child(obstacle_instance)
			connect_obstacle(obstacle_instance)
			
			var main_row = available_rows.pick_random()
			var current_row: int = main_row
			taken_rows.append(current_row)
			
			for row_num in obstacle_instance.Row_Count - 1:
				if obstacle_instance.direction_stretched == "Right":
					current_row += 1
				elif obstacle_instance.direction_stretched == "Left":
					current_row -= 1
				taken_rows.append(current_row)
			
			obstacle_instance.position.x = num * 3.0
			
			# Detect the number of walls and put it into the wall count variable
			if detect_walls(obstacle_instance):
				wall_count += 1
				wall_spawns.append(obstacle_instance)
	
	# If the wall count is 5 or more, pick a random one and delete it
	if wall_count >= 5 and not wall_spawns.is_empty():
		var to_free = wall_spawns.pick_random()
		if is_instance_valid(to_free):
			to_free.queue_free()


## Detects if the given obstacle is a wall
func detect_walls(obstacle: Obstacle):
	if obstacle.dodge_capabilities == "False":
		return true


## Finds the optimal distance for an obstacle
func find_optimal_distance() -> float:
	if not power:
		return 22.5

	var base_spd: float = power.base_speed if "base_speed" in power else 15.0
	var max_spd: float = power.max_speed if "max_speed" in power else 343.0
	var target_spd: float = power.target_speed if "target_speed" in power else base_spd

	var clamped_spd: float = clampf(target_spd, base_spd, max_spd)

	var base_rate: float = 1.0 / 1.50
	var max_rate: float = base_rate * 12.0

	var current_rate: float = remap(clamped_spd, base_spd, max_spd, base_rate, max_rate)
	return clamped_spd / current_rate

	roads.add_child(road_spawn)
	return road_spawn

	return side_spawn

	if body.is_in_group("Player"):

func _advance_row_queue() -> void:
	if current_rows.is_empty():
		return
	if is_instance_valid(row_to_free):
		row_to_free.queue_free()
	row_to_free = current_rows.pop_front()
	summon_row_obstacles()
	if not current_rows.is_empty():
		place_Area3D(current_rows.front().global_position.z)

func place_Area3D(z_loc: float) -> void:
	row_detector.global_position.z = z_loc

## Custom procedural 2D HUD overlay that renders dynamic radial cooldowns, active duration pie wedges, and armor segment shields beneath player indicators.
extends Node2D

@export_category("UI References & Offset")
## Reference to the main player node.
@export var player: Player

## Reference to the global power manager node tracking run states.
@export var power_node: Node

## Vertical offset in pixels relative to the player's invincibility indicator.
@export var vertical_offset: float = 105.0

## Vertical spacing between stacked gadget circles when multiple effects are active.
@export var circle_spacing: float = 105.0

## Default base radius of drawn UI circles in pixels.
var radius: float = 22.0

## Default fill color for active radial gauges.
const DEFAULT_FILL: Color = Color(0.20, 0.95, 0.35, 0.90)

## Divider line color used to separate power armor shield segments.
const DIVIDER_COLOR: Color = Color(0.08, 0.01, 0.08, 0.95)

## Color mapping table defining signature HUD colors for each gadget type.
const GADGET_COLORS: Dictionary = {
	"power_armor": Color(0.20, 0.95, 0.35, 0.90),
	"hoverboard": Color(1.00, 1.00, 1.00, 0.90),
	"magnet": Color(0.95, 0.25, 0.25, 0.90),
	"pulse_wave": Color(0.95, 0.20, 0.85, 0.90),
	"overdriven_rocket": Color(1.00, 0.55, 0.00, 0.90)
}

## Data container class representing rendering instructions for a single HUD status circle.
class CircleData:
	var fill_ratio: float = 0.0
	var is_thirds: bool = false
	var hits: int = 0
	var color: Color = Color.GREEN
	var bg_color: Color = Color.BLACK

## Active list of circle data entries to be rendered in the current frame draw call.
var active_circles: Array[CircleData] = []


func _ready() -> void:
	# High Z-index ensures UI elements render on top of world 3D geometry
	z_index = 100


func _process(_delta: float) -> void:
	if not power_node or not power_node.is_run_active:
		hide()
		return

	# Align HUD position and scale dynamically with player screen indicators
	if player and is_instance_valid(player.invincible_indicator):
		var inv = player.invincible_indicator
		global_position = inv.global_position + Vector2(0, vertical_offset)

		if inv.mesh:
			var max_dim: float = maxf(inv.mesh.get_aabb().size.x, inv.mesh.get_aabb().size.y)
			if max_dim > 0.0:
				radius = (max_dim * inv.scale.x) / 2.0
		elif inv.scale.x > 0:
			radius = 22.0 * inv.scale.x

	show()
	_update_circle_stack()
	queue_redraw()


## Queries active loadout and pickup nodes to construct rendering data for all active status rings.
func _update_circle_stack() -> void:
	active_circles.clear()
	if not player or not power_node:
		return

	var gadget = power_node.selected_gadget
	var equipped_id: String = gadget.gadget_id if gadget else ""
	var is_overcharged_state: bool = power_node.is_overcharged

	# Query temporary effect nodes attached to the player
	var active_hover = player.get_node_or_null("HoverboardInstance")
	var active_magnet = player.get_node_or_null("MagnetInstance")
	var active_armor = player.get_node_or_null("PowerArmorInstance")

	# 1. Process primary equipped loadout gadget indicator
	if gadget and equipped_id != "base_gadget":
		var primary_circle := CircleData.new()
		primary_circle.color = GADGET_COLORS.get(equipped_id, DEFAULT_FILL)
		primary_circle.bg_color = Color(primary_circle.color.r * 0.2, primary_circle.color.g * 0.2, primary_circle.color.b * 0.2, 0.65)

		var active_found: bool = false
		if equipped_id == "hoverboard" and active_hover:
			active_found = true
			primary_circle.fill_ratio = clampf(active_hover.duration / active_hover.max_duration, 0.0, 1.0)
		elif equipped_id == "magnet" and active_magnet:
			active_found = true
			primary_circle.fill_ratio = clampf(active_magnet.duration / active_magnet.max_duration, 0.0, 1.0)
		elif equipped_id == "power_armor" and active_armor:
			active_found = true
			primary_circle.is_thirds = true
			primary_circle.hits = active_armor.hits_remaining
		elif equipped_id == "overdriven_rocket" and is_overcharged_state:
			active_found = true
			primary_circle.fill_ratio = clampf(power_node.overdrive_rocket_timer / power_node.overdrive_rocket_max_duration, 0.0, 1.0)

		# Display cooldown recharge progress if no active effect duration is running
		if not active_found:
			primary_circle.fill_ratio = clampf(1.0 - power_node.gadget_cooldown_progress, 0.0, 1.0)

		active_circles.append(primary_circle)

	# 2. Append secondary indicators for temporary pickup effects unequipped in loadout
	if active_hover and equipped_id != "hoverboard":
		_add_circle("hoverboard", active_hover.duration / active_hover.max_duration)
	if active_magnet and equipped_id != "magnet":
		_add_circle("magnet", active_magnet.duration / active_magnet.max_duration)
	if active_armor and equipped_id != "power_armor":
		var c := _create_circle_data("power_armor")
		c.is_thirds = true
		c.hits = active_armor.hits_remaining
		active_circles.append(c)
	if is_overcharged_state and equipped_id != "overdriven_rocket":
		_add_circle("overdriven_rocket", power_node.overdrive_rocket_timer / power_node.overdrive_rocket_max_duration)


## Helper method to construct and append standard radial circle data entries.
func _add_circle(gadget_key: String, ratio: float) -> void:
	var c := _create_circle_data(gadget_key)
	c.fill_ratio = clampf(ratio, 0.0, 1.0)
	active_circles.append(c)


## Helper method initializing CircleData instances with signature color styling.
func _create_circle_data(gadget_key: String) -> CircleData:
	var c := CircleData.new()
	c.color = GADGET_COLORS.get(gadget_key, DEFAULT_FILL)
	c.bg_color = Color(c.color.r * 0.2, c.color.g * 0.2, c.color.b * 0.2, 0.65)
	return c


## Main CanvasItem draw callback that renders background rings, pie wedges, and armor segments.
func _draw() -> void:
	for i in range(active_circles.size()):
		var circle_data: CircleData = active_circles[i]
		var center: Vector2 = Vector2(0, i * circle_spacing)
		
		draw_circle(center, radius, circle_data.bg_color)
		if circle_data.is_thirds:
			_draw_armor_thirds(center, radius, circle_data.hits, circle_data.color)
		else:
			_draw_pie_wedge(center, radius, circle_data.fill_ratio, circle_data.color)


## Renders a filled pie wedge using polar coordinate trigonometry to represent radial progress.
func _draw_pie_wedge(center: Vector2, rad: float, ratio: float, color: Color) -> void:
	if ratio <= 0.001:
		return
	var points: PackedVector2Array = [center]
	var num_segments: int = max(4, int(ceil(32 * ratio)))
	var start_angle: float = -PI / 2.0  # Start pointing straight up (-90 degrees)
	var total_angle: float = ratio * TAU

	# Generate peripheral vertex points around circle circumference
	for i in range(num_segments + 1):
		var angle: float = start_angle + (float(i) / float(num_segments)) * total_angle
		points.append(center + Vector2(cos(angle), sin(angle)) * rad)
	draw_polygon(points, PackedColorArray([color]))


## Renders up to 3 equal 120-degree arc segments to represent remaining power armor hits.
func _draw_armor_thirds(center: Vector2, rad: float, hits: int, color: Color) -> void:
	var segment_angle: float = TAU / 3.0  # 120 degrees per segment in radians
	var start_offset: float = -PI / 2.0  # Align top segment centered straight up

	# Draw filled arc polygons for active hit points
	for i in range(hits):
		var points: PackedVector2Array = [center]
		var seg_start: float = start_offset + (i * segment_angle)
		for j in range(11):
			var angle: float = seg_start + (float(j) / 10.0) * segment_angle
			points.append(center + Vector2(cos(angle), sin(angle)) * rad)
		draw_polygon(points, PackedColorArray([color]))

	# Draw segment divider lines across shield boundaries
	for i in range(3):
		var div_angle: float = start_offset + (i * segment_angle)
		var div_end: Vector2 = center + Vector2(cos(div_angle), sin(div_angle)) * rad
		draw_line(center, div_end, DIVIDER_COLOR, 3.0)

@export var power_node: Node

## Onready reference to the gadget dropdown selector menu.
@onready var gadget_option: OptionButton = %GadgetOption

## Onready reference to the run start confirmation button.
@onready var start_button: Button = %StartButton

## Registry array of dictionaries mapping display names to GadgetBase string IDs.
const GADGET_OPTIONS: Array[Dictionary] = [
	{"name": "None (Standard Run)", "id": ""},
	{"name": "Pulse Wave", "id": "pulse_wave"},
	{"name": "Magnet", "id": "magnet"},
	{"name": "Power Armor", "id": "power_armor"},
	{"name": "Hoverboard", "id": "hoverboard"},
	{"name": "Overdriven Rocket", "id": "overdriven_rocket"}
]


func _ready() -> void:
	# Enable visible mouse cursor for menu interaction
	Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
	if has_node("CanvasLayer"):
		$CanvasLayer.show()
	show()

	# Configure element dimensions and disable keyboard focus traps
	gadget_option.custom_minimum_size = Vector2(250, 40)
	start_button.custom_minimum_size = Vector2(250, 40)
	start_button.focus_mode = Control.FOCUS_NONE
	gadget_option.focus_mode = Control.FOCUS_NONE

	# Populate OptionButton menu items dynamically from registry dictionary
	gadget_option.clear()
	for i in range(GADGET_OPTIONS.size()):
		gadget_option.add_item(GADGET_OPTIONS[i]["name"], i)

	start_button.pressed.connect(_on_start_button_pressed)


## Confirms loadout selection, instantiates gadget via factory, and begins the run.
func _on_start_button_pressed() -> void:
	start_button.release_focus()
	if not power_node:
		return

	# Clean up previous gadget instance to prevent memory leaks across consecutive runs
	if power_node.selected_gadget:
		power_node.selected_gadget.queue_free()
		power_node.selected_gadget = null

	# Lookup selected gadget ID from OptionButton dropdown state
	var selected_idx: int = gadget_option.get_selected_id()
	var gadget_id: String = GADGET_OPTIONS[selected_idx]["id"]
	
	# Instantiate gadget using the GadgetBase factory method if selected
	if not gadget_id.is_empty():
		var gadget_instance: GadgetBase = GadgetBase.create_by_id(gadget_id)
		if gadget_instance:
			power_node.add_child(gadget_instance)
			power_node.selected_gadget = gadget_instance

	# Capture mouse cursor for gameplay controls, hide UI overlay, and start run
	Input.mouse_mode = Input.MOUSE_MODE_CAPTURED
	if has_node("CanvasLayer"):
		$CanvasLayer.hide()
	hide()
	power_node.start_run()
